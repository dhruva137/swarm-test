\begin{frame}{The question}
\begin{itemize}\setlength{\itemsep}{0.7em}
\item Overseeing a swarm means linking actions under different identities to one effort.
\item This operation is called \hl{linkage}.
\item Where does a real swarm's coordination signal live?
\item Where does oversight lose it?
\end{itemize}
\end{frame}

\begin{frame}{Setup}
\small
\begin{itemize}\setlength{\itemsep}{0.45em}
\item \textbf{Real data.} collusion.wiki: 14,591 edits, 3,102 self-chosen names. SwarmTraces: 189,579 attack artifacts.
\item \textbf{Monitors.} Retrieval, gradient-boosted classifier, Qwen3 judges (neutral vs.\ collusion-framed prompts), linear probes, causal steering.
\item \textbf{Controlled swarms.} 12 agents, 5 rounds, told vs.\ not told about a content monitor.
\item \textbf{Ground truth.} Shared uncommon URL; URLs deleted from monitor inputs; time-block splits.
\end{itemize}
\end{frame}

\begin{frame}{Takeaways}
\small
{\normalsize Oversight loses the signal wherever it reads a surface that agents can change (names, metadata, how coordination sounds), and keeps it where it reads what agents say or the monitor's representation of it.\par}
\vspace{0.8em}
\begin{itemize}\setlength{\itemsep}{0.45em}
\item \textbf{Infrastructure.} Keep and index content for cross-agent retrieval; issue platform identities.
\item \textbf{Monitors.} Score linkage with a linear probe or calibrated similarity, not framed yes/no verdicts.
\item \textbf{Judges.} Red-team with the swarm's own language: one relay sentence adds up to +1.9 log-odds.
\end{itemize}
\end{frame}
